\section*{Idea}
Un solo lazo: el RP2350 conmuta el MOSFET (PWM = corriente de fusi\'on + patr\'on de
orde\~no) y \textbf{detecta todo} de $R=V/I$. El sensado NO usa chips: $V$ de la junta por
un \textbf{divisor} de resistencias al ADC, e $I$ por un \textbf{shunt} (resistencia de
derivaci\'on) al ADC. La pieza queda a GND (seguro) y el sensado vale siempre.

\section*{C\'omo DETECTA (todo de $R=V/I$, dos ADC)}
$R = \dfrac{V_{\text{ADC0}}\cdot\frac{R_1+R_2}{R_2}}{\,V_{\text{ADC1}}/R_{sh}\,}$\quad
(el RP2350 muestrea en la ventana ON, que \'el mismo controla).

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Lo que ve} & \textbf{Se\~nal} & \textbf{Decisi\'on}\\
\midrule
$R=\infty$ ($I\approx0$) & alambre no toca & avanza hasta tocar\\
$R$ cae a $\sim$\SI{20}{m\ohm} & \textbf{toc\'o} (corto fr\'io) & guarda $R_{\text{fr\'io}}$ (auto-cal)\\
$R$ sube $\sim7\times$ & calienta (TCR) & $T=25+\frac{R/R_{fr}-1}{\alpha}$ (term\'ometro)\\
$R/R_{fr}>6$ & $\approx$ liquidus & FUNDE, guarda $R_{\min}$\\
$R>4\,R_{\min}$ & \textbf{cuello} & baja a $I_{\text{fondo}}$ (corte suave)\\
$R\to\infty$ & gota soltada & avanza, repite\\
\bottomrule
\end{tabular}
\end{center}

\section*{Lista (sensado 100\,\% pasivo)}
\begin{itemize}\setlength\itemsep{1pt}
\item \textbf{Sensado:} $R_1,R_2$ (divisor de $V$) $+$ $R_{sh}$ (shunt de $I$) $+$ 2 diodos clamp a \SI{3.3}{V}. \emph{Cero chips de sensado.}
\item \textbf{Potencia:} Q1 (N-MOSFET \SIrange{60}{100}{V}), $D_{fw}$ Schottky $\sim$\SI{130}{A}, $L$ y $C$ de la SOMI, gate driver (1 IC \emph{o} t\'otem de 2 transistores).
\item \textbf{Reusado:} impresora (XYZ + avance + cama caliente) + fuente \SI{24}{V}.
\item \textbf{Respaldo opcional:} comparador LM393 sobre el umbral de cuello $\to$ corte r\'apido al pin SD del driver.
\end{itemize}
