// Cuaternión del marco MAGNÉTICO en t  (lo usan lentes 1,2,5)
function magQuat(t, P){
  const rot = 2*Math.PI*t/P;
  const qy = new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(0,1,0), rot);
  const qx = new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(1,0,0), TILT);
  return qy.multiply(qx);            // Ry·Rx
}
// Eje de ROTACIÓN: simplemente Ry(rot) — lentes 3,4 (toro X, jets, viento)

/* ═══════════ SISTEMA DE LENTES: pesos w_i(viewMode) ═══════════
   viewMode ∈ [0,5).  Ventana triangular suave por lente con SOLAPE 0.5
   (cross-fade) → corte óptico motivado, nunca dos lentes "encimadas duro". */
function lensWeights(v){
  const tri = (c)=> Math.max(0, 1 - Math.abs(v - c)/1.0);   // pico 1 en c, 0 a ±1
  const s = (x)=> x*x*(3-2*x);                                // smoothstep
  return {
    surface: s(tri(0)),   // 1: SUPERFICIE lensada (NICER)
    mag:     s(tri(1)),   // 2: MAGNÉTICA (dipolo + R_LC)
    xray:    s(tri(2)),   // 3: RAYOS X (toro + jets eje ROTACIÓN)
    gamma:   s(tri(3)),   // 4: GAMMA (viento de pares)
    radio:   s(tri(4)),   // 5: RADIO (haces eje MAGNÉTICO = el pulso)
  };
}

const SURFACE_FRAG = /* glsl */`
precision highp float;
varying vec2 vUv;
uniform vec3 uCamPos,uCamFwd,uCamRight,uCamUp;
uniform float uTanHalfFov,uAspect,uRot,uTilt,uRS,uRSTAR,uW;  // uW = peso de la lente
mat3 rotY(float a){float c=cos(a),s=sin(a);return mat3(c,0,s,0,1,0,-s,0,c);}
mat3 rotX(float a){float c=cos(a),s=sin(a);return mat3(1,0,0,0,c,-s,0,s,c);}
float cap(vec3 n,vec3 c,float cr){return smoothstep(cr,mix(cr,1.0,0.45),dot(n,normalize(c)));}
// mapa NICER J0030 en marco MAGNÉTICO local (antes de tilt/rot)
float nicer(vec3 nLocal){
  float spot = cap(nLocal, vec3(0.60,0.77,0.22), cos(radians(8.0)));      // mancha
  float cre=0.0;                                                          // media luna
  cre=max(cre,cap(nLocal,vec3(-0.10,0.82,0.565),cos(radians(6.5))));
  cre=max(cre,cap(nLocal,vec3(-0.287,0.82,0.497),cos(radians(6.5))));
  cre=max(cre,cap(nLocal,vec3(-0.44,0.82,0.369),cos(radians(6.5))));
  return max(spot,cre);
}
void main(){
  if(uW<0.001){gl_FragColor=vec4(0.0);return;}
  vec2 uv=vUv*2.0-1.0; uv.x*=uAspect;
  vec3 rd=normalize(uCamFwd+uv.x*uTanHalfFov*uCamRight+uv.y*uTanHalfFov*uCamUp);
  vec3 rp=uCamPos;
  float dt=0.05*uRSTAR; vec3 col=vec3(0.0); bool hit=false; vec3 hitN;
  // raymarch geodésico Schwarzschild — parada INVERTIDA en la esfera estelar
  for(int i=0;i<160;i++){
    float r=length(rp);
    if(r<uRSTAR){ hit=true; hitN=normalize(rp); break; }   // ← INTERSECTA superficie
    if(r>length(uCamPos)+20.0) break;
    vec3 toC=-rp/r; float strength=2.0*uRS/(r*r);          // MISMA ley que BH
    rd=normalize(rd+toC*strength*dt);
    dt=max(0.02*uRSTAR, (r-uRSTAR)*0.25);                  // pasos finos cerca de la superficie
    rp+=rd*dt;
  }
  if(hit){
    // n en marco MAGNÉTICO local: deshacer rot y tilt
    vec3 nLocal = rotX(-uTilt)*rotY(-uRot)*hitN;
    float h = nicer(nLocal);
    // Beloborodov: angulo de emision real. -rd = dir hacia el observador.
    float cosPsi = clamp(dot(hitN, -rd),-1.0,1.0);          // psi = normal vs vista
    float u = uRS/uRSTAR;                                    // 0.326
    float cosAlpha = (cosPsi - u)/(1.0-u);                   // emision local
    float vis = smoothstep(-0.15, 0.05, cosAlpha);          // hasta α~100° aún emite
    // cuerpo negro 10^6 K = blanco-azul PALIDO (RGB 0.34/0.51/1.0)
    vec3 hot  = vec3(0.62,0.78,1.0);
    vec3 crust= vec3(0.018,0.025,0.050);                    // corteza casi negra (no gris mate!)
    // LIMBO que BRILLA: el hemisferio trasero enroscado se acumula en el borde
    float limb = pow(1.0 - max(cosPsi,0.0), 1.3);           // brillante en el borde
    vec3 c = crust + hot*pow(h,0.7)*1.9*vis + hot*0.35*limb*vis;
    col = c;
  }
  gl_FragColor = vec4(col*uW, uW);    // additive; el cuerpo manda forma con bloom
}`;

// renderAt(t) escribe: timeRef (cámara), animTime, y viewMode(t):
//   La narración dicta los cortes. 5 ventanas iguales + cross-fade de 1.2s:
function viewModeAt(t, DUR){
  // 5 segmentos; centro de cada lente en t = (i+0.5)*DUR/5
  return Math.min(4.999, (t/DUR)*5.0 - 0.5 + 0.5);  // = (t/DUR)*5; lensWeights cross-fadea
}
