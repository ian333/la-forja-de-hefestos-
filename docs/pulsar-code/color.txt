// (a1) COLOR CUERPO-NEGRO — Planck real para la NS. GLSL drop-in en
//      CinematicPulsar.tsx → NeutronStar() fragmentShader. Reemplaza el bloque
//      crust/hot/col actual (líneas ~190-194). FÍSICA: T>5e5 K = cola
//      Rayleigh-Jeans (Wien peak en rayos X). El crust REAL casi no emite
//      visible → casi negro; los hot spots revientan a BLANCO-AZUL PÁLIDO.
// ───────────────────────────────────────────────────────────────────────────
//  Constantes (valores EXACTOS calculados Planck→sRGB, normalizados a luma):
//    TINT_HOT (paleness 0.55) = vec3(0.685, 0.747, 1.000)   // hot spot pálido
//    TINT_CRUST (chroma pura)  = vec3(0.301, 0.439, 1.000)   // SOLO para el tinte
//                                                               del residual, MUY tenue
//  uniform float uTcrust;  // 0..1 normalizado: J0030=0.50 (1e6K), Vela=0.65 (1.5e6K)

  // ── Color cuerpo-negro: crust CASI NEGRO en visible (XUV peak), hot spots reventando ──
  // El crust de una NS a 1e6 K emite el grueso de su flujo en rayos X; en banda
  // VISIBLE es la cola Rayleigh-Jeans → tenue. NO pintar un globo azul.
  const vec3 TINT_HOT   = vec3(0.685, 0.747, 1.000);  // Planck 1e6K, paleness 0.55
  const vec3 TINT_CRUST = vec3(0.301, 0.439, 1.000);  // chroma pura (solo tinte)
  // crust: emisión visible MÍNIMA (Rayleigh-Jeans) + leve realce polar térmico.
  // base 0.006 (casi negro), no 0.014/0.020/0.045 cobalto del actual.
  float pole = pow(abs(n.y), 5.0);
  vec3 crust = TINT_CRUST * (0.010 + 0.022 * pole);        // tenue, frío, NO cobalto
  // hot spots: intensidad emisiva ALTA (1e6 K es brutal) → ACES+bloom los lleva
  // a blanco-azul pálido CONSERVANDO la forma (pow 0.8 = borde nítido del cap).
  vec3 hot = TINT_HOT * 5.2;                               // HDR: revienta vía bloom
  vec3 col = crust + hot * pow(h, 0.8);
  // limbo: light-bending hace BRILLAR el borde (3/4 de superficie visible). El
  // limbo recibe un realce azul-pálido suave (foto-esfera doblada al borde).
  float limb = pow(max(dot(normalize(vNview), vView), 0.0), 0.45);
  col *= mix(0.18, 1.0, limb);                             // 0.18 = crust casi negro al centro
  col += TINT_HOT * 0.35 * pow(1.0 - limb, 2.0) * smoothstep(0.0,0.4,h+0.15); // glo de limbo
  gl_FragColor = vec4(col, 1.0);

// ───────────────────────────────────────────────────────────────────────────

//      Bloom DOMADO que conserva la media luna. CA=0 (confeti).

    bloomThreshold: 0.65,     // ALTO: solo los hot spots cruzan (crust no brilla)
    bloomRadius: 0.55,        // corto: halo ceñido = conserva la media luna
    dofFocusDistance: 0.02,
    dofFocalLength: 0.05,
    dofBokehScale: 0,         // NS = sphere sólida; DOF plano lava todo (off)

    grainOpacity: 0.03,
    vignetteDarkness: 0.9,    // void NEGRO en esquinas (como el BH)

  },

// ───────────────────────────────────────────────────────────────────────────
// (c) STARFIELD REAL — reemplaza el <Starfield> de points en CinematicPulsar.
//     OPCIÓN ELEGIDA: portar el starfield GLSL del BHRaytraced.stars() a un quad
//     fullscreen propio (cáscara que SÍ llena el cuadro, magnitud pow, color B-V,
//     Vía Láctea, nebulosa Cangrejo etiquetada). Justificación vs points: los
//     points no entran en cuadro como contexto y no tienen magnitud física; el
//     enfoque GLSL es el MISMO que da el fondo de la vara BH. CA=0.
//     Color B-V real (Planck→sRGB calculado): O=#6c8cff B=#96adff A=#d4d7ff
//     F=#ffe6e2 G=#ffd2b0 K/M=#ffa65d.
// ───────────────────────────────────────────────────────────────────────────
function Starfield({ crab = false }: { crab?: boolean }) {
  const { camera, size } = useThree();
  const mat = useMemo(() => new THREE.ShaderMaterial({
    depthWrite: false, depthTest: false, toneMapped: false,
    uniforms: {
      uCamFwd:{value:new THREE.Vector3()}, uCamRight:{value:new THREE.Vector3()},
      uCamUp:{value:new THREE.Vector3()}, uTanHalfFov:{value:0.4}, uAspect:{value:1.0},
      uCrab:{value: crab?1.0:0.0},
    },
    vertexShader:`varying vec2 vUv; void main(){vUv=uv; gl_Position=vec4(position.xy,0.999,1.0);}`,
    fragmentShader:/* glsl */`
      precision highp float; varying vec2 vUv;
      uniform vec3 uCamFwd,uCamRight,uCamUp; uniform float uTanHalfFov,uAspect,uCrab;
      float h21(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
      float n2(vec2 p){vec2 i=floor(p),f=fract(p);float a=h21(i),b=h21(i+vec2(1,0)),
        c=h21(i+vec2(0,1)),d=h21(i+vec2(1,1));vec2 u=f*f*(3.-2.*f);
        return mix(mix(a,b,u.x),mix(c,d,u.x),u.y);}
      float fbm(vec2 p){float v=0.,a=.5;for(int i=0;i<5;i++){v+=a*n2(p);p*=2.07;a*=.5;}return v;}
      // B-V chromaticity LUT (Planck→sRGB real): O→M sequence
      vec3 bvColor(float t){ // t in 0..1 = blue..red
        vec3 O=vec3(0.42,0.55,1.00), A=vec3(0.83,0.85,1.00),
             G=vec3(1.00,0.90,0.79), M=vec3(1.00,0.65,0.37);
        if(t<0.45) return mix(O,A,t/0.45);
        if(t<0.75) return mix(A,G,(t-0.45)/0.30);
        return mix(G,M,(t-0.75)/0.25);
      }
      vec3 stars(vec3 dir){
        dir=normalize(dir);
        float theta=atan(dir.z,dir.x), phi=asin(clamp(dir.y,-1.,1.));
        vec3 col=vec3(0.);
        for(int layer=0;layer<4;layer++){
          float scale=340.*pow(1.55,float(layer));
          vec2 uvS=vec2(theta*scale,phi*scale*2.)+vec2(float(layer)*4.1,float(layer)*7.7);
          vec2 cell=floor(uvS), cf=fract(uvS);
          for(int dx=-1;dx<=1;dx++)for(int dy=-1;dy<=1;dy++){
            vec2 cc=cell+vec2(float(dx),float(dy));
            float hh=h21(cc+float(layer)*11.3);
            float thr=0.965-0.005*float(layer);
            if(hh>thr){
              vec2 pp=vec2(float(dx)+h21(cc+2.),float(dy)+h21(cc+5.));
              vec2 dd=cf-pp; float d2=dot(dd,dd);
              // MAGNITUD: pow alto → pocas estrellas dominan (ley de magnitud real)
              float mag=pow(hh-thr,1.6);
              float inten=exp(-d2*80.)*mag*620.;
              vec3 tint=bvColor(h21(cc+7.0));            // color B-V real por estrella
              col+=tint*inten*(1.0/pow(1.3,float(layer)));
            }
          }
        }
        // Vía Láctea sutil (fix anti-morado del BH: gris-frío tenue, NO lavanda)
        float band=exp(-dir.y*dir.y*4.5);
        float dust=fbm(vec2(theta*6.,phi*4.))*band;
        col+=vec3(0.34,0.36,0.42)*dust*0.08;
        col+=vec3(0.80,0.78,0.70)*pow(band,1.6)*0.04;
        float darkB=smoothstep(0.3,0.55,fbm(vec2(theta*5.,phi*3.)))*band;
        col*=1.0-darkB*0.4;
        // NEBULOSA CANGREJO (etiquetada: solo lentes X/gamma, uCrab=1). SN 1054,
        // filamentos + synchrotron azul-blanco. Evocativa → se ETIQUETA en caption.
        if(uCrab>0.5){
          vec3 cd=normalize(vec3(0.5,0.25,0.83));        // posición fija en el cielo
          float ang=acos(clamp(dot(dir,cd),-1.,1.));
          float neb=exp(-pow(ang/0.18,2.0));
          float fil=fbm(vec2(theta*9.+ang*20.,phi*9.));
          col+=vec3(0.55,0.78,1.00)*neb*(0.20+0.45*fil); // synchrotron azul-blanco
          col+=vec3(1.00,0.45,0.42)*neb*fil*0.12;         // filamentos Hα tenues
        }
        return col;
      }
      void main(){
        vec2 uv=vUv*2.-1.; uv.x*=uAspect;
        vec3 rd=normalize(uCamFwd+uv.x*uTanHalfFov*uCamRight+uv.y*uTanHalfFov*uCamUp);
        gl_FragColor=vec4(stars(rd)*0.9,1.0);            // CA=0 (cero confeti)
      }`,
  }), [crab]);
  useFrame(() => {
    const fwd=new THREE.Vector3(); camera.getWorldDirection(fwd);
    const right=new THREE.Vector3().crossVectors(fwd,camera.up).normalize();
    const up=new THREE.Vector3().crossVectors(right,fwd).normalize();
    mat.uniforms.uCamFwd.value.copy(fwd); mat.uniforms.uCamRight.value.copy(right);
    mat.uniforms.uCamUp.value.copy(up);
    mat.uniforms.uTanHalfFov.value=Math.tan(((camera as THREE.PerspectiveCamera).fov*Math.PI/180)/2);
    mat.uniforms.uAspect.value=size.width/size.height;
  });
  return <mesh renderOrder={-100} frustumCulled={false} material={mat}>
    <planeGeometry args={[2,2]} /></mesh>;
}
