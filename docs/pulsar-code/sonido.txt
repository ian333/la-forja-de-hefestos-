NÚCLEO DE LA FÍSICA (replica del shader):
  def pole_axis(t, ps):
      rot = 2*pi*t/ps.P; st = sin(ps.tilt)
      return stack([sin(rot)*st, cos(ps.tilt), cos(rot)*st])   # = three.js Ry·Rx·ŷ (verificado match=True)
  def to_cam_dir(t, ps):
      azim = ps.az0 + (t/ps.dur)*ps.az_sweep; cp = cos(ps.incl)   # az0=-0.6, sweep=1.4 (= cameraProgram)
      v = stack([cp*sin(azim), sin(ps.incl), cp*cos(azim)]); return v/norm(v)
  def signed_dot(t, ps):  return sum(pole_axis*to_cam_dir)        # +1 polo N, -1 polo S (interpulso)
  def pulse_curve(t, ps, pole_sign):  a=clip(pole_sign*signed_dot,0,1); return clip((a-0.6)/0.4,0,1)**2.4
  def find_pulse_onsets(...): cruces de subida de pulse>=0.12  → ataque de cada beep
