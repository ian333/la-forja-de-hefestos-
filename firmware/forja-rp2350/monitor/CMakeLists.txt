cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(forja_mon C CXX ASM)
set(PICO_BOARD pico2)
pico_sdk_init()
add_executable(forja_mon main.c)
target_link_libraries(forja_mon pico_stdlib hardware_adc)
pico_enable_stdio_usb(forja_mon 1)
pico_enable_stdio_uart(forja_mon 0)
pico_add_extra_outputs(forja_mon)
