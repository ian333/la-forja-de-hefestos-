cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(forja_boost C CXX ASM)
set(PICO_BOARD pico2)
pico_sdk_init()
add_executable(forja_boost main.c)
target_link_libraries(forja_boost pico_stdlib hardware_adc hardware_pwm)
pico_enable_stdio_usb(forja_boost 1)
pico_enable_stdio_uart(forja_boost 0)
pico_add_extra_outputs(forja_boost)
